## Overview

Create the entire database layer: Drizzle config, DB singleton client, 4 schema files (split by domain), barrel index, and run initial migration.

## Key Insights

- `better-sqlite3` is **synchronous** — no `async/await` on any DB call
- `globalThis._db` singleton prevents multiple connections during Next.js dev hot-reload
- Enable **WAL** pragma for concurrent reads (multiple Next.js requests)
- Enable **foreign_keys** pragma (SQLite disables FK by default)
- PostgreSQL `UUID` → SQLite `text('id').primaryKey().$defaultFn(() => crypto.randomUUID())`
- PostgreSQL `JSONB`/arrays → SQLite `text({ mode: 'json' })` (Drizzle handles parse/stringify)
- PostgreSQL `DECIMAL(5,1)` → SQLite `real`
- No `SERIAL` — use `integer` with `autoincrement` only where needed (not PKs)
- `timestamps_list[]` (PostgreSQL array) → `text({ mode: 'json' }).$type<string[]>()`

## Requirements

## Architecture

```
drizzle.config.ts              ← points to schema glob + sqlite DB file
src/lib/db/
├── index.ts                   ← singleton: Database + drizzle() + pragmas
├── schema/
│   ├── index.ts               ← re-export all tables + relations
│   ├── projects.ts            ← projects, tasks, monthly_targets
│   ← seo.ts                   ← seo_results, keyword_rankings
│   ├── members.ts             ← members, salary_payments
│   └── sync-logs.ts           ← sync_logs
└── migrations/                ← generated by drizzle-kit (do not edit)
```

### Step 3 — Create `src/lib/db/index.ts`

```ts
import Database from 'better-sqlite3';
import { drizzle } from 'drizzle-orm/better-sqlite3';
import * as schema from './schema';
import path from 'path';

const DB_PATH = path.join(process.cwd(), 'data', 'seo-manager.db');

declare global {
  // eslint-disable-next-line no-var
  var _db: ReturnType<typeof drizzle> | undefined;
}

function createDb() {
  const sqlite = new Database(DB_PATH);
  // Enable WAL for concurrent reads
  sqlite.pragma('journal_mode = WAL');
  // Enable foreign key enforcement
  sqlite.pragma('foreign_keys = ON');
  return drizzle(sqlite, { schema });
}

export const db = globalThis._db ?? (globalThis._db = createDb());
```

### Step 4 — Create `src/lib/db/schema/projects.ts`

```ts
import { sqliteTable, text, integer, real, uniqueIndex } from 'drizzle-orm/sqlite-core';
import { sql } from 'drizzle-orm';

export const projects = sqliteTable('projects', {
  id: text('id').primaryKey().$defaultFn(() => crypto.randomUUID()),
  name: text('name').notNull(),
  sheetId: text('sheet_id').notNull(),
  sheetName: text('sheet_name').notNull(),
  monthlyTarget: integer('monthly_target').notNull().default(20),
  rankingSheetUrl: text('ranking_sheet_url'),
  createdAt: text('created_at').notNull().default(sql`(datetime('now'))`),
});

export const tasks = sqliteTable('tasks', {
  id: text('id').primaryKey().$defaultFn(() => crypto.randomUUID()),
  projectId: text('project_id').notNull().references(() => projects.id, { onDelete: 'cascade' }),
  stt: integer('stt').notNull().default(0),
  year: integer('year').notNull(),
  month: integer('month').notNull(),
  parentKeyword: text('parent_keyword').notNull().default(''),
  keywordSub: text('keyword_sub').notNull().default(''),
  keywordCount: integer('keyword_count').notNull().default(0),
  keywordsList: text('keywords_list', { mode: 'json' }).$type<string[]>().notNull().default(sql`'[]'`),
  searchVolume: integer('search_volume').notNull().default(0),
  title: text('title').notNull().default(''),
  outline: text('outline').notNull().default(''),
  timelineOutline: text('timeline_outline').notNull().default(''),
  statusOutline: text('status_outline').notNull().default(''),
  pic: text('pic').notNull().default(''),
  contentFile: text('content_file').notNull().default(''),
  deadline: text('deadline'),       // ISO date string YYYY-MM-DD or null
  statusContent: text('status_content').notNull().default(''),
  linkPublish: text('link_publish').notNull().default(''),
  publishDate: text('publish_date'), // ISO date string or null
  note: text('note').notNull().default(''),
  monthYear: text('month_year').notNull().default(''),
  createdAt: text('created_at').notNull().default(sql`(datetime('now'))`),
  updatedAt: text('updated_at').notNull().default(sql`(datetime('now'))`),
});

export const monthlyTargets = sqliteTable('monthly_targets', {
  id: text('id').primaryKey().$defaultFn(() => crypto.randomUUID()),
  projectId: text('project_id').notNull().references(() => projects.id, { onDelete: 'cascade' }),
  month: integer('month').notNull(),
  year: integer('year').notNull(),
  target: integer('target').notNull().default(20),
}, (t) => ({
  uniqueProjectMonthYear: uniqueIndex('monthly_targets_project_month_year_idx')
    .on(t.projectId, t.month, t.year),
}));
```

## Risk Assessment

| Risk | Mitigation |
|------|------------|
| `crypto.randomUUID()` not available in Node <18 | Node 18+ required; Next.js 14 requires it anyway |
| `better-sqlite3` native build fails | Already in package.json, should be compiled |
| WAL mode not compatible with some filesystems | Only affects network/NFS drives; local dev is fine |
